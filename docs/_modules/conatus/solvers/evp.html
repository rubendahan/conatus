"""Generalized eigenvalue problems on a discretization plan.

Convention (``conatus.ir.problem.EVP``): the eigenvalue sigma appears in the LHS,

    (L + sigma M) x = 0        i.e.   -L x = sigma M x,

group by group. Every mode carries its relative backward residual
``||(L + sigma M) x|| / ((||L|| + |sigma| ||M||) ||x||)``. Nothing is dropped
silently: infinite eigenvalues (M singular on tau and boundary rows) are
counted, and spurious finite modes of the tau method (Gardner et al. 1989 for
Orr–Sommerfeld) are only removed by an explicit resolution-drift test
(Boyd 2001), whose outcome is part of the result.
"""
from __future__ import annotations

import numbers
from collections.abc import Callable, Sequence
from dataclasses import dataclass
from dataclasses import field as dc_field
from typing import Any

import numpy as np
import scipy.linalg as sl
import scipy.sparse as sp
import scipy.sparse.linalg as spla

from .._deprecation import renames
from ..tolerances import REQUIRED, require

__all__ = ["EVPResult", "solve_dense", "solve_sparse", "drift_filter", "sensitivity",
           "matrix_derivative", "biorthogonality", "check_drift_tol", "cluster_sensitivity",
           "eigen_scale", "balance_pencil"]

INF_TOL = 1e3 * np.finfo(float).eps     # |beta| <= INF_TOL |alpha| (scaled) -> infinite eigenvalue
# ARPACK stopping tolerance of shift-invert (declared default): 0 = machine precision, the
# strictest; every returned mode carries its residual (conatus.tolerances.DECLARED).
ARNOLDI_TOL = 0.0
#: Shift-invert with a target on an eigenvalue (to SHIFT_GUARD of max(|target|, eigen_scale)): the
#: Arnoldi shift is moved by that distance in the direction SHIFT_GUARD_DIRECTION (off the real
#: axis, so a real spectrum is not hit again). sqrt(eps): far enough for max|nu| to stay below
#: 1/(sqrt(eps) unit), near enough that the requested modes are still the nearest.
SHIFT_GUARD = 1e-5
SHIFT_GUARD_DIRECTION = complex(0.6, 0.8)
DRIFT_WHY = ("the drift tolerance decides which modes are kept: a mode is returned only if it "
             "reappears at the finer resolution within it (relative to |sigma|, floored at "
             "DRIFT_FLOOR times the median |sigma| of the finer spectrum); 1e-6 "
             "keeps resolved modes and removes spurious tau modes in the usual cases, a tighter "
             "value also removes marginally resolved physical modes")


#: Floor of the drift ratio, as a fraction of the median |sigma| of the finer spectrum (audit
#: invariance). The ratio was |dsigma| / max(|sigma|, 1): the 1 is in the unit of the
#: eigenvalue, so the same modes written in another time unit (sigma / 1e4) were judged on an
#: absolute drift and 9 under-resolved Orr-Sommerfeld modes passed instead of 1. The median of
#: the finer spectrum is in the unit of sigma (a handful of spurious modes do not move it); a mode
#: below DRIFT_FLOOR of it is "near zero" for this spectrum and its drift is measured against
#: that floor (a neutral mode computed as rounding noise at both resolutions stays kept).
DRIFT_FLOOR = 1e-3


def drift_ratios(lo: np.ndarray, hi: np.ndarray) -> np.ndarray:
    """|sigma_lo - nearest sigma_hi| / max(|sigma_lo|, DRIFT_FLOOR median |sigma_hi|) for each
    eigenvalue of ``lo``: invariant under a change of unit of sigma and under any scaling of the
    equations (audit invariance)."""
    from .._scales import spectral_scale
    lo, hi = np.asarray(lo), np.asarray(hi)
    # the scale of the finer spectrum: the median of its nonzero moduli (critic review: the plain
    # median is 0 when most eigenvalues are 0, and the rounding-noise neutral modes, compared with
    # a floor of 0, were dropped as drifting); the largest when fewer than half are nonzero
    floor = DRIFT_FLOOR * spectral_scale(hi) if hi.size else 0.0
    return np.array([np.min(np.abs(hi - s)) / max(abs(s), floor, np.finfo(float).tiny)
                     for s in lo])


def check_drift_tol(tol: Any, where: str, name: str = "drift_tol") -> float:
    """``drift_tol`` given and meaningful: a finite real
    number in (0, DRIFT_TOL_MAX]. Zero or less keeps nothing; above DRIFT_TOL_MAX (1e-2) most
    modes, spurious ones included, find a neighbour at the finer resolution, so the test no
    longer filters (``drift_tol=1.0`` was accepted and looked filtered)."""
    import numbers

    from ..tolerances import DRIFT_TOL_MAX
    tol = require(tol, name, where=where, suggest=f"{name}=1e-6", why=DRIFT_WHY)
    if (isinstance(tol, bool) or not isinstance(tol, numbers.Real) or not np.isfinite(tol)
            or not 0 < tol <= DRIFT_TOL_MAX):
        from ..errors import OptionError
        raise OptionError(f"{where}: {name}={tol!r} must be a number in (0, {DRIFT_TOL_MAX:g}]",
                          why="drift_tol is the relative distance within which a mode must be found "
                              "again at the finer resolution: 0 keeps nothing, and above "
                              f"{DRIFT_TOL_MAX:g} (conatus.tolerances.DRIFT_TOL_MAX) most modes, "
                              "spurious tau modes included, find a neighbour, so nothing is "
                              "filtered while the result reads as filtered",
                          fix=f"{name}=1e-6 (resolved modes kept, tau modes removed in the "
                              "usual cases)")
    return float(tol)


def _norm1(A) -> float:
    if sp.issparse(A):
        return float(abs(A).sum(axis=0).max()) if A.nnz else 0.0
    return float(np.linalg.norm(A, 1))

@dataclass
class EVPResult:
    """Eigenvalues sigma of (L + sigma M) x = 0 for one group, with residuals and a filter log."""

    values: np.ndarray
    vectors: np.ndarray | None               # right eigenvectors (columns), or None
    residuals: np.ndarray                     # relative backward residual per mode
    method: str
    label: str = ""
    n_infinite: int = 0
    left: np.ndarray | None = None            # left eigenvectors y: y^H (L + sigma M) = 0
    drift: np.ndarray | None = None           # drift ratio per mode (if a drift test ran)
    log: list[str] = dc_field(default_factory=list)
    # (plan, group, reduced, {variable name: unknown name}) of the group that produced the
    # vectors, set by conatus.eigenvalues(plan, ...): what mode(i) needs to unpack them
    context: Any = dc_field(default=None, repr=False, compare=False)
    # growth convention of the eigenvalue: its name, the rule
    # ('real', 'imag' or a callable of sigma; None = not declared) and the sentence saying why;
    # set by conatus.eigenvalues(plan, ...) from the problem, used by order() and explain()
    eigenvalue: str = "sigma"
    growth: Any = dc_field(default=None, repr=False, compare=False)
    growth_note: str = ""
    # spurious-mode filter: None when the problem has no tau (nothing
    # to filter), else a sentence: the drift test that ran, or that none ran
    filter_note: str | None = None
    # spectral-tail spurious-mode flag (conatus.solvers.spurious):
    # relative tail energy per mode and the flag (None: the test did not run), and its sentence
    tails: np.ndarray | None = dc_field(default=None, repr=False, compare=False)
    spurious: np.ndarray | None = dc_field(default=None, repr=False, compare=False)
    spurious_note: str = ""
    # resolution of the selected mode (
    # conatus.solvers.spurious.check_selected): "" or the sentence saying that order()[0] of a
    # result without a drift test has a tail above UNRESOLVED_TAIL_TOL
    unresolved_note: str = ""
    # moduli |alpha/beta| of the pairs removed as infinite only because M is numerically singular
    # (singular values of the row-equilibrated M below n eps sigma_max, no exactly zero row or column
    # behind them): in floating point they cannot be told from finite eigenvalues of that size;
    # empty when every removed pair is proven infinite (beta = 0, or a zero
    # row/column of M)
    infinite_ambiguous: np.ndarray = dc_field(default_factory=lambda: np.zeros(0), repr=False,
                                              compare=False)

    @property
    def flagged(self) -> np.ndarray:
        """Indices of the modes flagged spurious by the spectral-tail test (empty if none, or if
        the test did not run: see ``spurious_note``)."""
        if self.spurious is None:
            return np.zeros(0, dtype=int)
        return np.nonzero(self.spurious)[0]

    @property
    def unresolved(self) -> np.ndarray:
        """Indices of the modes that pass the spectral-tail flag but whose tail exceeds the
        stricter ``conatus.tolerances.UNRESOLVED_TAIL_TOL`` (under-resolved physical modes). Empty if the tail test did not run. Kept and ranked as any mode: the result
        says it (``unresolved_note``) when the selected mode ``order()[0]`` is one of them."""
        if self.tails is None:
            return np.zeros(0, dtype=int)
        from ..tolerances import UNRESOLVED_TAIL_TOL
        ok = np.ones(self.values.size, bool) if self.spurious is None else ~self.spurious
        return np.nonzero(ok & (self.tails > UNRESOLVED_TAIL_TOL))[0]

    def order(self, key: str = "growth") -> np.ndarray:
        """Indices from the least to the most stable mode by the declared growth convention of
        the eigenvalue (``key='growth'``: Re s for exp(s t), Im c for a phase speed c;
        an undeclared convention ranks by Re, said by :meth:`growth_rule`), or by
        decreasing real part (``'real'``) or imaginary part (``'imag'``).

        Modes flagged spurious by the spectral-tail test (:attr:`flagged`) come
        after every mode that passes it (each part ranked by ``key``): ``order()[0]`` is the
        least stable mode that passes; the result keeps one index per mode."""
        if key == "growth":
            k = -self._growth_fn()(self.values)
        elif key == "real":
            k = -self.values.real
        elif key == "imag":
            k = -self.values.imag
        else:
            from ..errors import OptionError, did_you_mean
            raise OptionError(f"order(key={key!r})", fix="key='growth' (declared growth "
                              "convention), 'real' (decreasing real part) or 'imag' (decreasing "
                              "imaginary part)" + did_you_mean(str(key), ("growth", "real", "imag")))
        if self.spurious is None or not self.spurious.any():
            return np.argsort(k, kind="stable")
        return np.lexsort((k, self.spurious.astype(int)))

def _residuals(L, M, w, V) -> np.ndarray:
    """Relative backward residual ||L x + w M x|| / ((||L|| + |w| ||M||) ||x||). When ||L|| +
    |w| ||M|| vanishes (w = 0 in a group where L = 0: the k = 0 mode of a periodic advection-
    diffusion operator) the scale is ||M|| instead: the residual of such a mode was 0/0 = NaN
    (Y3); M = 0 is refused before any solve."""
    nL, nM = _norm1(L), _norm1(M)
    R = L @ V + (M @ V) * w[None, :]
    scale = nL + np.abs(w) * nM
    scale = np.where(scale > 0, scale, nM)
    return np.linalg.norm(R, axis=0) / (scale * np.linalg.norm(V, axis=0))


def _require_mass(M, L=None) -> None:
    """M = 0 means the eigenvalue never enters the problem: refuse rather than return []. With
    ``L``: both square, of one shape, and finite (audit inputs: a shape mismatch or a
    NaN entry was a bare ValueError of SciPy)."""
    from ..errors import EquationError, InputTypeError
    if L is not None:
        if L.ndim != 2 or L.shape != M.shape or L.shape[0] != L.shape[1]:
            raise InputTypeError(f"pencil (L, M): shapes {L.shape} and {M.shape}, two square "
                                 "matrices of one shape are needed",
                                 fix="L and M of the same group (plan.matrices(group))")
        data = [A.data if sp.issparse(A) else np.asarray(A) for A in (L, M)]
        if not all(np.all(np.isfinite(d)) for d in data):
            raise InputTypeError("pencil (L, M): non-finite entries (NaN or inf)",
                                 why="eigenvalues of a NaN matrix mean nothing",
                                 fix="check the parameters and coefficients the matrices were "
                                     "built with")
    if _norm1(M) == 0.0:
        raise EquationError("M is identically zero: the eigenvalue appears in no equation "
                            "(see EVP.check())")


#: independent points of the regularity test of L + s M (times ||L||/||M||). A regular pencil is
#: singular at its eigenvalues, so one singular sample proves nothing.
PENCIL_SHIFTS = (0.3719 + 0.6153j, -0.5827 + 0.2391j, 0.1643 - 0.8761j)


def pencil_sigma_ratio(L: np.ndarray, M: np.ndarray, shift: complex) -> float:
    """sigma_min / sigma_max of L + s M at s = shift ||L||/||M||, rows and columns equilibrated
    (two sweeps), so that bad scaling is not read as singularity."""
    nL, nM = _norm1(L), _norm1(M)
    A = np.asarray(L + shift * (nL / max(nM, 1e-300) if nL > 0 else 1.0) * M, dtype=complex)
    for _ in range(2):
        A = A / np.maximum(np.abs(A).max(axis=1, keepdims=True), 1e-300)
        A = A / np.maximum(np.abs(A).max(axis=0, keepdims=True), 1e-300)
    sv = np.linalg.svd(A, compute_uv=False)
    return float(sv[-1] / sv[0]) if sv.size and sv[0] > 0 else 0.0


def numerically_singular_pencil(L: np.ndarray, M: np.ndarray, tol: float) -> tuple[bool, float]:
    """(L + s M rank-deficient at every shift of PENCIL_SHIFTS, the largest ratio). The rank is
    numerical: the answer holds to the tolerance ``tol`` on sigma_min / sigma_max."""
    best = 0.0
    for s in PENCIL_SHIFTS:
        best = max(best, pencil_sigma_ratio(L, M, s))
        if best >= tol:
            return False, best
    return True, best


def refuse_singular_pencil(L: np.ndarray, M: np.ndarray, a: np.ndarray, b: np.ndarray,
                           label: str = "") -> None:
    """Refuse a pencil whose determinant vanishes for every s.

    A QZ pair with alpha and beta both at rounding level of the pencil is indeterminate, neither a
    finite nor an infinite eigenvalue. Such a pair is only a candidate: a regular pencil near
    singularity can show one, so the sampled rank of L + s M decides. A regular pencil with
    infinite eigenvalues has beta = 0 and alpha != 0, and is not concerned."""
    from ..errors import RankError
    n = a.size
    eps = np.finfo(float).eps
    size = max(float(np.abs(a).max(initial=0.0)), float(np.abs(b).max(initial=0.0)))
    if not n or not np.any((np.abs(a) <= 100 * n * eps * size) & (np.abs(b) <= 100 * n * eps * size)):
        return
    singular, ratio = numerically_singular_pencil(L, M, 10 * n * eps)
    if singular:
        raise RankError(f"pencil L + s M{f' [{label}]' if label else ''} is singular: "
                        f"sigma_min/sigma_max <= {ratio:.1e} at {len(PENCIL_SHIFTS)} independent s",
                        why="its determinant vanishes for every s, so it has no discrete spectrum: "
                            "QZ returns an indeterminate pair alpha = beta = 0, and the other "
                            "pairs depend on rounding",
                        fix="an unknown is undetermined or two equations are dependent: add the "
                            "missing condition or gauge")


def _row_max(A, mask=None) -> np.ndarray:
    """Largest |entry| of each row of A (dense or sparse), over the columns of ``mask`` if given."""
    if sp.issparse(A):
        A = abs(sp.csr_matrix(A))
        if mask is not None:
            A = A[:, np.flatnonzero(mask)]
        return np.asarray(A.max(axis=1).toarray()).ravel() if A.shape[1] else np.zeros(A.shape[0])
    A = np.abs(np.asarray(A))
    if mask is not None:
        A = A[:, mask]
    return A.max(axis=1, initial=0.0)


def _col_nonzero(A) -> np.ndarray:
    return (np.asarray(abs(sp.csc_matrix(A)).max(axis=0).toarray()).ravel() > 0 if sp.issparse(A)
            else np.asarray(A).any(axis=0))


def eigen_scale(L, M) -> float:
    """Unit of the eigenvalue of L + s M: the smallest, over the rows where M has an entry, of
    (largest |L| entry over the columns where M has an entry) / (largest |M| entry).

    The tau and boundary rows (no M) and the tau columns (no M) are left out, so the number scales
    as the eigenvalue does: exactly by c when the physical rows of L are multiplied by c (a change of
    time unit), by 1/H^2 for a diffusion operator whose coordinate is stretched by H, and not at all
    when an equation is multiplied by a constant (the ratio is taken row by row). 1.0 when no row
    gives a ratio (L = 0 on those rows)."""
    Q = _col_nonzero(M)
    mq = _row_max(M)
    P = mq > 0
    lq = _row_max(L, Q)
    rho = lq[P] / mq[P]
    rho = rho[(rho > 0) & np.isfinite(rho)]
    # the smallest ratio: the slowest physical row sets the unit, so the rule (a row of M
    # below n eps of its L row, after balancing, is numerically zero) now reads "an eigenvalue more
    # than 1/(n eps) times the slowest scale cannot be told from infinity". A row of M at rounding
    # level (1e-17 where the exact M has a zero row, a huge ratio) never sets the unit, even
    # beside a genuine 1e13 row (test_f7: a median there made the noise row count as mass)
    return float(rho.min()) if rho.size else 1.0


#: Scale factors of :func:`balance_pencil` within [1/BALANCE_FREE, BALANCE_FREE] are not applied,
#: and larger ones only beyond it: a pencil already in units near its own is solved as in 0.3.0,
#: bit for bit. Measured (r031e): applying every factor moved a marginal index-2 Jordan pair at
#: infinity of pipe flow (m = 0) across the n eps rank line, where it came back as a finite pair
#: +-8e12; with 1e4 a residual imbalance cost up to 3e-9 relative on s u = c u'' (Neumann, SI
#: lengths) and 5e-10 on a random pencil; with 1e2 (1e4 on the lift columns, which move such
#: Jordan pairs most) the worst of 42 scalings is 1.6e-11.
BALANCE_FREE = 1e2


def _beyond(f: float, free: float = BALANCE_FREE) -> float:
    """The part of a scale factor beyond [1/free, free] (1.0 inside it)."""
    return float(f / min(max(f, 1.0 / free), free))


def balance_factors(L, M, free: float = BALANCE_FREE) -> tuple[float, np.ndarray, np.ndarray]:
    """(mu, dc, r) of :func:`balance_pencil`, for a dense or sparse pencil; ``free``: an eigenvalue
    unit within [1/free, free] is not applied, nor a lift-column factor within [1/free^2, free^2]
    (1.0: every factor). The lift columns get the wider band: a factor of 127 on them was enough
    to move a marginal Jordan pair at infinity of the pipe-flow tutorial (m = k = 1) from removed
    to kept as an ambiguous finite 2.6e13."""
    n = L.shape[0]
    def beyond(f: float) -> float:
        return _beyond(f, free)
    mu = beyond(eigen_scale(L, M))
    aL = abs(sp.csc_matrix(L)) if sp.issparse(L) else sp.csc_matrix(np.abs(np.asarray(L)))
    aM = abs(sp.csc_matrix(M)) if sp.issparse(M) else sp.csc_matrix(np.abs(np.asarray(M)))
    aL.eliminate_zeros()
    aM.eliminate_zeros()
    P = np.diff(aM.tocsr().indptr) > 0              # rows with an M entry
    Q = np.diff(aM.indptr) > 0                      # columns with an M entry
    dc = np.ones(n)
    if P.any() and (~Q).any():
        # typical size of each row: its largest entry over the columns with M (and mu M on the
        # rows with M); rows without any such entry (a gauge on a Lagrange unknown) do not count
        t = _row_max(aL, Q)
        t[P] = np.maximum(t[P], mu * _row_max(aM)[P])
        # only the columns that enter the rows with M alone (tau and lift unknowns); a column that
        # also enters a row without M (pressure, in the continuity row) is left as given
        outside = np.diff(sp.csc_matrix(aL[np.flatnonzero(~P)]).indptr) > 0 if (~P).any() \
            else np.zeros(n, bool)
        lift = np.flatnonzero(~Q & ~outside)
        if lift.size:
            tt = np.where(t > 0, t, np.inf)
            sub = sp.diags(1.0 / tt) @ aL[:, lift]
            v = np.asarray(sub.max(axis=0).toarray()).ravel()
            dc[lift] = [_beyond(x, free * free) if x > 0 else 1.0 for x in v]
    rL = _row_max(aL @ sp.diags(1.0 / dc))
    rM = mu * _row_max(aM @ sp.diags(1.0 / dc))
    r = np.maximum(rL, rM)
    return mu, dc, np.where(r > 0, r, 1.0)


def balance_pencil(Ld: np.ndarray, Md: np.ndarray, free: float = BALANCE_FREE):
    """(Le, Me, mu, dc, r): the dense pencil L + s M written as R^-1 (L + (mu sigma) M) C^-1, with
    s = mu sigma, Le = R^-1 L C^-1 and Me = mu R^-1 M C^-1, every row of [Le, Me] of largest entry 1.

    QZ is backward stable in the norm of the whole pencil. Rows are equilibrated (audit invariance: an equation written times 1e-8 lost 8 digits), but with the eigenvalue in its own
    unit a physical row of size kappa/H^2 sat beside O(1) tau and boundary rows: s T = kappa T'' in
    SI units lost 3 digits (kappa = 1.4e-7, H = 4 km), and with s = c u'' at c = 1e16 the M rows fell
    below n eps of the L rows and every finite mode was removed as infinite (hunt).
    Three steps, so that the balanced pencil is the same, to rounding, after multiplying an
    equation by a constant, rescaling the eigenvalue, or stretching a coordinate: mu =
    eigen_scale(L, M) puts the eigenvalue in the unit of the slowest physical row; each column
    without M that enters only the rows with M (a tau or lift unknown, whose size is free) is
    divided by its largest entry relative to the typical size of those rows (their largest entry
    on the columns with M, and mu M), so it follows the eigenvalue unit; each row is then divided by
    its largest entry. A column without M that also enters a row without M (pressure, in the
    continuity row) is left as given. Each factor is applied only beyond BALANCE_FREE
    (:func:`_beyond`), so a pencil in units near its own is solved as in 0.3.0. The columns that
    carry M are left as given (their equilibration compounds the range of M on Chebyshev
    operators). Right vectors of L + s M are C^-1 y = y / dc; left vectors are R^-1 z = z / r."""
    mu, dc, r = balance_factors(Ld, Md, free)
    r = r[:, None]
    return Ld / dc / r, (mu * Md) / dc / r, mu, dc, r

def solve_dense(L, M, *, left: bool = False, label: str = "") -> EVPResult:
    """All eigenvalues by the QZ algorithm (scipy.linalg.eig) on the balanced pencil
    (:func:`balance_pencil`), infinite ones counted and removed."""
    Ld = L.toarray() if sp.issparse(L) else np.asarray(L)
    Md = M.toarray() if sp.issparse(M) else np.asarray(M)
    _require_mass(Md, Ld)
    Le, Me, mu, dc, r = balance_pencil(Ld, Md)
    out = sl.eig(-Le, Me, left=left, right=True, homogeneous_eigvals=True)
    (a, b), *vecs = out
    if left:
        VL, VR = vecs
        VL = VL / r            # y^H (L + s M) = 0  <=>  (D^-1 y)^H D (L + s M) = 0, D = 1/r
    else:
        VL, VR = None, vecs[0]
    VR = VR / dc[:, None]      # (L + s M) C^-1 y = 0  <=>  x = C^-1 y
    refuse_singular_pencil(Le, Me, a, b, label)
    # the finite/infinite split on the pencil balanced with every factor (free = 1): QZ ran with
    # the factors beyond BALANCE_FREE only, but the split must not depend on the units at all
    # (a mass row at rounding level beside its L row is judged in the unit of the slowest row)
    Lf, Mf, muf, _dcf, _rf = balance_pencil(Ld, Md, free=1.0)
    finite, note, amb, _kept = _finite_eigenvalues(a * (mu / muf), b, Lf, Mf, Md, unit=muf)
    a = a * mu                 # sigma = alpha / beta in the unit mu; s = mu sigma
    if mu != 1.0:
        note += f"; QZ ran on s / {mu:.3e} (balance_pencil)"
    w = a[finite] / b[finite]
    V = VR[:, finite]
    res = _residuals(Ld, Md, w, V)
    r = EVPResult(w, V, res, "dense QZ (scipy.linalg.eig)", label, int((~finite).sum()),
                  VL[:, finite] if left else None)
    r.infinite_ambiguous = amb
    r.log.append(note)
    return r




def _finite_eigenvalues(a: np.ndarray, b: np.ndarray, Le: np.ndarray, Me: np.ndarray,
                        Md: np.ndarray, unit: float = 1.0):
    """(mask of the finite eigenvalues alpha/beta of the QZ pairs, the sentence for the log, the
    moduli of the pairs removed as infinite on numerical rank alone, the moduli of the pairs kept
    as finite although they are ambiguous).

    A pair is "under the threshold" when beta = 0 exactly, or when |beta| <= INF_TOL |alpha|
    after scaling by ||L|| / ||M|| (rows equilibrated). A regular pencil has at least
    n - rank(M) infinite eigenvalues (more with Jordan blocks at infinity) and none when M is
    nonsingular, so the threshold alone does not decide (earlier versions: L = -diag(1, 1e13),
    M = I lost its exact eigenvalue 1e13 as infinite).

    The numerical rank of M counts the singular values of Me above n eps sigma_max(Me), where Me
    is M with each row divided by the largest entry of that row in L or M (the rows of the pencil
    equilibrated together, as for QZ). D (L + s M) has the eigenvalues of L + s M for every positive
    diagonal D and gives the same Me, so the rank no longer depends on the scale of each equation.
    Two earlier versions did: the rank of M as given
    depended on the scale of each equation, so L = -I, M = diag(1, 1e-17) lost its
    root 1e17 while the same pencil with its second row times 1e17 kept it; the rank of M with each
    row divided by its own largest M entry turned a row of rounding noise
    (1e-17 ||M|| where the exact M has a zero row) into a row of order one. A row of M that is
    below n eps of the L entries of its row is numerically zero in either scaling; the pair it
    removes is reported as ambiguous (a finite eigenvalue that large cannot be told from infinity
    in double precision), with its modulus, in the log and in ``EVPResult.infinite_ambiguous``.
    Pairs with beta = 0, or removed by an exactly zero row or column of M, are infinite.

    With M of full rank every pair with beta != 0 is finite; the pairs under the threshold are
    named in the log. With M singular, a pair is infinite when beta = 0, when it is one of the
    n - rank(M) pairs of smallest |beta| / |alpha| under the threshold, or when its scaled
    |beta| / |alpha| is below n eps (the rank tolerance). The other pairs under the threshold
    (|beta| / |alpha| between n eps and INF_TOL) are kept as finite and named as ambiguous with
    their modulus. When fewer pairs are under the threshold than n - rank(M), the pairs of
    smallest |beta| / |alpha| that the rank says should be infinite are kept and named too.
    ``unit``: the eigenvalue unit of the balanced pencil (alpha / beta times ``unit`` is the
    eigenvalue); the moduli in the sentence and in the returned arrays are in the caller's unit."""
    n = a.size
    eps = np.finfo(float).eps
    scale = max(_norm1(Le), 1e-300) / max(_norm1(Me), 1e-300)
    ratio = np.abs(b) * scale / np.maximum(np.abs(a), np.finfo(float).tiny)
    under = (b == 0) | (np.abs(b) * scale <= INF_TOL * np.abs(a))
    sv = sl.svdvals(Me) if n else np.zeros(0)
    rank_tol = n * eps
    rank = int(np.count_nonzero(sv > rank_tol * sv[0])) if n and sv[0] > 0 else 0
    deficiency = n - rank
    # null directions of M that are exact: zero rows or zero columns of M as given
    structural = max(int(np.count_nonzero(~Md.any(axis=1))), int(np.count_nonzero(~Md.any(axis=0)))) if n else 0
    rule = (f"|beta| <= {INF_TOL:.1e} |alpha| after scaling by ||L||/||M||, rows equilibrated; "
            f"M has numerical rank {rank} of {n} (singular values of M with the rows of the pencil "
            "equilibrated, above n eps sigma_max)")
    none = np.zeros(0)
    if deficiency == 0:
        infinite = b == 0
        kept = under & ~infinite
        note = (f"{int(infinite.sum())} infinite eigenvalue(s) removed (beta = 0); M is "
                f"nonsingular ({rule}), so no other pair is infinite"
                + (f": {int(kept.sum())} pair(s) under the threshold kept as finite eigenvalues "
                   f"(|sigma| >= {unit * np.abs(a[kept] / b[kept]).min():.3e})" if kept.any() else ""))
        return ~infinite, note, none, none
    order = np.argsort(np.where(under, ratio, np.inf), kind="stable")
    by_rank = np.zeros(n, bool)
    by_rank[order[:min(deficiency, int(under.sum()))]] = True
    infinite = (b == 0) | by_rank | (under & (ratio <= rank_tol))
    note = f"{int(infinite.sum())} infinite eigenvalue(s) removed ({rule})"
    # pairs removed on numerical rank alone: beyond the exact null directions, and beta != 0
    soft = infinite & (b != 0)
    n_soft = max(0, min(int(soft.sum()), int(infinite.sum()) - structural))
    amb_inf = none
    if n_soft:
        idx = np.nonzero(soft)[0]
        # an exact zero row gives beta at rounding level (the largest moduli): the smallest moduli
        # are the ones only the numerical rank removed
        idx = idx[np.argsort(np.abs(a[idx] / b[idx]))][:n_soft]
        amb_inf = unit * np.sort(np.abs(a[idx] / b[idx]))
        note += (f"; {n_soft} of them removed on numerical rank alone (M has {deficiency} "
                 f"direction(s) below n eps, {structural} exact zero row(s)/column(s)): ambiguous, "
                 "a finite eigenvalue of modulus "
                 f"{amb_inf.min():.3e} .. {amb_inf.max():.3e} cannot be told from an infinite one "
                 "in double precision (EVPResult.infinite_ambiguous)")
    amb = np.nonzero(under & ~infinite)[0]
    kept = none
    if amb.size:
        sizes = unit * np.abs(a[amb] / b[amb])
        kept = np.sort(sizes)
        note += (f"; {amb.size} pair(s) under the threshold beyond n - rank(M) = {deficiency}, with "
                 f"|beta|/|alpha| above n eps = {rank_tol:.1e}, are ambiguous (finite eigenvalues, "
                 "or a Jordan block at infinity) and are kept as finite eigenvalues of modulus "
                 f"{sizes.min():.3e} .. {sizes.max():.3e}")
    short = deficiency - int(under.sum())
    if short > 0:
        rest = np.argsort(np.where(infinite, np.inf, ratio), kind="stable")[:short]
        rest = rest[~infinite[rest]]
        if rest.size:
            sizes = unit * np.abs(a[rest] / b[rest])
            kept = np.sort(np.concatenate([kept, sizes]))
            note += (f"; M has {deficiency} null direction(s) but only {int(under.sum())} pair(s) "
                     f"are under the threshold: the {rest.size} pair(s) of smallest |beta|/|alpha| "
                     f"(modulus {sizes.min():.3e} .. {sizes.max():.3e}) are ambiguous and kept as "
                     "finite eigenvalues")
    return ~infinite, note, amb_inf, kept

@renames("conatus.solvers.evp:solve_sparse")
def solve_sparse(L, M, target: complex, n_modes: int = 6, *, label: str = "",
                 tol: float = ARNOLDI_TOL) -> EVPResult:
    """``n_modes`` eigenvalues nearest ``target`` by shift-invert Arnoldi on (L + target M)^{-1} M
    (``k=`` is the deprecated former name of ``n_modes``: k is a wavenumber everywhere else).

    Works for singular, non-symmetric M (no M-inner product assumed): the
    standard problem  -(L + s M)^{-1} M x = nu x  has nu = 1 / (sigma - s).
    """
    from ..errors import OptionError, check_count
    L = sp.csc_matrix(L, dtype=complex)
    M = sp.csc_matrix(M, dtype=complex)
    _require_mass(M, L)
    n = L.shape[0]
    # target: a finite number; n_modes: an integer the Arnoldi method can return (audit inputs,
    # target=nan factorised a NaN matrix, "0.2" was a bare UFuncTypeError, and
    # n_modes was clipped to n - 2 while the log said the requested count)
    check_target(target)
    check_count(n_modes, "n_modes", "shift-invert")
    if n_modes > n - 2:
        raise OptionError(f"shift-invert: n_modes={n_modes} > {n - 2} (= size - 2, the most the "
                          f"Arnoldi method returns for a pencil of size {n})",
                          fix=f"n_modes <= {n - 2}, or method='dense' for every mode")
    # target on an eigenvalue (hunt): L + target M is singular (SuperLU raised a bare
    # RuntimeError at target = 0 beside a neutral mode), or so nearly singular that nu of that mode
    # is ~1e15 and every other mode fell under the INF_TOL max|nu| test and was dropped as
    # infinite. The shift is then moved by GUARD (relative to max(|target|, eigen_scale)), off the
    # real axis; the eigenvalues returned are those of L + s M, nearest the target as asked.
    unit = max(abs(complex(target)), eigen_scale(L, M))
    step = SHIFT_GUARD * unit * SHIFT_GUARD_DIRECTION
    shift, moved = complex(target), ""
    # the balanced pencil of solve_dense (hunt): R^-1 (L + s M) C^-1 with the rows
    # and the lift columns scaled and the eigenvalue in the unit mu; nu = 1/(sigma - shift/mu)
    mu, dc, rr = balance_factors(L, M)
    Ls = (sp.diags(1.0 / rr) @ L @ sp.diags(1.0 / dc)).tocsc()
    Ms = (sp.diags(mu / rr) @ M @ sp.diags(1.0 / dc)).tocsc()
    for attempt in range(2):
        try:
            lu = spla.splu((Ls + (shift / mu) * Ms).tocsc())
            op = spla.LinearOperator((n, n), matvec=lambda x, lu=lu: -lu.solve(Ms @ x),
                                     dtype=complex)
            nu, V = spla.eigs(op, k=n_modes, which="LM", tol=tol)
            nu = nu / mu                      # 1/(s - shift) in the caller's unit
            V = V / dc[:, None]
        except RuntimeError as exc:          # "Factor is exactly singular"
            if attempt:
                from ..errors import RankError
                raise RankError(f"shift-invert: L + s M is singular at s = {target} and at "
                                f"s = {shift}", why="the pencil is singular for every s (an "
                                "unknown is undetermined)", fix="method='dense' names the "
                                "singular pencil; add the missing condition or gauge") from exc
            shift, moved = complex(target) + step, "factorisation exactly singular"
            continue
        if attempt or np.abs(nu).max(initial=0.0) * abs(step) <= 1.0:
            break
        shift, moved = complex(target) + step, "an eigenvalue within the guard distance"
    # nu = 0 <=> sigma = infinity (M singular): counted, never returned as a finite mode.
    scale = max(np.abs(nu).max(initial=0.0), 1e-300)
    keep = np.abs(nu) > INF_TOL * scale
    w = shift + 1.0 / nu[keep]
    V = V[:, keep]
    res = _residuals(L, M, w, V)
    n_inf = int((~keep).sum())
    r = EVPResult(w, V, res, f"shift-invert Arnoldi (scipy eigs), target {target}", label, n_inf)
    r.log.append(f"ARPACK tol = {tol:g}" + (" (machine precision, declared default ARNOLDI_TOL)"
                                             if tol == 0 else " (given)"))
    if moved:
        r.log.append(f"target {target}: {moved}; Arnoldi ran at the shift {shift:.6g} "
                     f"(target + SHIFT_GUARD {SHIFT_GUARD:.1e} x {unit:.3e}); the eigenvalues "
                     "are those of L + s M")
    r.log.append(f"{n_modes} modes requested nearest {target}; {n_inf} infinite eigenvalues "
                 f"(|nu| <= {INF_TOL:.1e} max|nu|) removed")
    if w.size < n_modes:
        import warnings

        from ..errors import CorrectnessWarning
        msg = (f"shift-invert: {w.size} of the {n_modes} modes requested nearest {target} are "
               f"returned; {n_inf} Ritz value(s) nu = 1/(s - shift) were at or below "
               f"INF_TOL max|nu| = {INF_TOL * scale:.1e}: infinite eigenvalues (M singular on the "
               "tau and boundary rows), or finite ones too far from the shift to tell apart; "
               "method='dense' returns the whole finite spectrum")
        r.log.append(msg)
        warnings.warn(msg, CorrectnessWarning, stacklevel=3)
    return r




def drift_filter(lo: EVPResult, hi: EVPResult, tol: float = REQUIRED) -> EVPResult:
    """Keep the modes of ``lo`` that reappear in ``hi`` (a finer resolution) within ``tol``.

    ``tol`` has no default: it decides which modes are returned.

    Drift ratio = |sigma_lo - nearest sigma_hi| / max(|sigma_lo|, DRIFT_FLOOR median |sigma_hi|)
    (:func:`drift_ratios`: no unit of sigma enters). Spurious tau modes move by
    orders of magnitude between resolutions; resolved physical modes do not. An under-resolved
    physical mode also moves: a removed mode is not converged by this test, not necessarily spurious.
    """
    tol = check_drift_tol(tol, "drift_filter()", "tol")      # given and in range
    if hi.values.size == 0:
        raise ValueError("drift test needs a non-empty high-resolution spectrum")
    d = drift_ratios(lo.values, hi.values)
    keep = np.nonzero(d <= tol)[0]
    out = lo.select(keep, f"resolution drift <= {tol:.0e} against {hi.values.size} modes at the "
                          f"finer resolution; {lo.values.size - keep.size} unresolved or spurious "
                          "(tau) modes removed")
    out.drift = d[keep]
    return out


def biorthogonality(res: EVPResult, M, idx: Sequence[int]) -> np.ndarray:
    """Matrix y_i^H M x_j over the chosen modes (diagonal-dominant for distinct eigenvalues)."""
    if res.left is None:
        raise ValueError("left eigenvectors not computed (solve_dense(..., left=True))")
    Y = res.left[:, idx]
    X = res.vectors[:, idx]
    return Y.conj().T @ (M @ X)


def sensitivity(res: EVPResult, i: int, M, dL, dM=None) -> complex:
    """d sigma / dp for mode i from its left and right eigenvectors.

    From (L + sigma M) x = 0 and y^H (L + sigma M) = 0:
        d sigma/dp = - y^H (dL/dp + sigma dM/dp) x / (y^H M x).
    """
    if res.left is None:
        raise ValueError("left eigenvectors not computed (solve_dense(..., left=True))")
    # a multiple eigenvalue (hunt 5D): inside the eigenspace QZ returns unrelated
    # left and right bases, and -y_i^H dL x_i / y_i^H M x_i matched no branch (the crossing
    # L = -(I + p B), B = [[1, 2], [2, -1]] gave 1, -1 for the Rellich slopes +-sqrt(5)). A
    # multiple eigenvalue has no single derivative: refused, with the slopes of its branches
    cluster = multiple_eigenvalue(res, i)
    if cluster.size > 1:
        from ..errors import SpectralError
        try:
            slopes = cluster_sensitivity(res, cluster, M, dL, dM)
            known = (f"the branches through it have the slopes {np.array2string(slopes, precision=8)} "
                     "(eigenvalues of the derivative projected on the eigenspace, "
                     "cluster_sensitivity)")
        except SpectralError as exc:
            known = str(exc).split("\n")[0]
        raise SpectralError(f"sensitivity: eigenvalue {complex(res.values[i]):.10g} (mode {i}) is "
                            f"multiple (modes {cluster.tolist()} within MULTIPLE_TOL = "
                            f"{MULTIPLE_TOL:g} of it): it has no single derivative; {known}",
                            fix="solvers.evp.cluster_sensitivity(res, idx, M, dL, dM) for the "
                                "slopes of the branches")
    x, y, s = res.vectors[:, i], res.left[:, i], res.values[i]
    num = y.conj() @ (dL @ x)
    if dM is not None:
        num = num + s * (y.conj() @ (dM @ x))
    return complex(-num / (y.conj() @ (M @ x)))


#: Two eigenvalues closer than MULTIPLE_TOL times max(|sigma|, spectral scale of the result) are
#: one multiple eigenvalue for :func:`sensitivity` (QZ separates a double eigenvalue by about
#: sqrt(eps) when it is defective, by rounding when it is semisimple).
MULTIPLE_TOL = 1e-6
#: Y^H M X of a multiple eigenvalue with sigma_min below DEFECTIVE_TOL times ||Y|| ||M X|| is
#: singular: the eigenvalue is defective and :func:`cluster_sensitivity` refuses it.
DEFECTIVE_TOL = 1e-8


def multiple_eigenvalue(res: EVPResult, i: int) -> np.ndarray:
    """Indices of the modes of ``res`` within MULTIPLE_TOL of mode ``i`` (``i`` included)."""
    from .._scales import spectral_scale
    w = np.asarray(res.values)
    unit = max(abs(w[i]), spectral_scale(w), np.finfo(float).tiny)
    return np.flatnonzero(np.abs(w - w[i]) <= MULTIPLE_TOL * unit)


def cluster_sensitivity(res: EVPResult, idx: Sequence[int], M, dL, dM=None) -> np.ndarray:
    """Slopes d sigma / dp of the branches through a multiple eigenvalue (modes ``idx``).

    For a semisimple eigenvalue sigma with right basis X and left basis Y (Y^H M X invertible),
    the branches of L(p) + s M(p) through it have the slopes eig(-(Y^H M X)^-1 Y^H (dL + sigma dM) X)
    (first-order perturbation of a semisimple eigenvalue; for a simple one this is
    :func:`sensitivity`). A defective eigenvalue (Y^H M X singular) has no slopes: its branches go
    as a fractional power of dp, refused."""
    from ..errors import SpectralError
    if res.left is None:
        raise ValueError("left eigenvectors not computed (solve_dense(..., left=True))")
    idx = np.asarray(idx, dtype=int)
    X, Y = res.vectors[:, idx], res.left[:, idx]
    s = complex(np.mean(np.asarray(res.values)[idx]))
    G = Y.conj().T @ (M @ X)
    sv = np.linalg.svd(G, compute_uv=False)
    scale = np.linalg.norm(Y, axis=0).max() * np.linalg.norm(M @ X, axis=0).max()
    if not sv.size or sv[-1] <= DEFECTIVE_TOL * scale:
        raise SpectralError(f"eigenvalue {s:.10g} of multiplicity {idx.size} is defective "
                            f"(Y^H M X has sigma_min/scale {sv[-1] / scale if sv.size else 0:.1e})"
                            ": its branches are not differentiable in p (Puiseux series)",
                            fix="perturb the problem off the degeneracy, or follow the branches "
                                "with ParametricEVP.track")
    D = Y.conj().T @ (dL @ X)
    if dM is not None:
        D = D + s * (Y.conj().T @ (dM @ X))
    return np.linalg.eigvals(-np.linalg.solve(G, D))


def matrix_derivative(build: Callable[[float], tuple], p: float, rel_h: float = 1e-3):
    """(dL/dp, dM/dp) by Richardson-extrapolated central differences (error O(h^4)).

    ``build(p) -> (L, M)`` must keep the sparsity layout fixed (same plan structure).
    """
    h = rel_h * max(abs(p), 1.0)

    def cd(hh):
        Lp, Mp = build(p + hh)
        Lm, Mm = build(p - hh)
        return (Lp - Lm) / (2 * hh), (Mp - Mm) / (2 * hh)

    L1, M1 = cd(h)
    L2, M2 = cd(h / 2)
    return (4 * L2 - L1) / 3, (4 * M2 - M1) / 3
