"""Symmetry subspaces: declaration and parity propagation.

 is IR only: a ``SymmetrySubspace`` is generated by reflections (Z2^k); each
field has a parity per generator; every node propagates parities; problems check
that equations are equivariant and boundary conditions compatible. The adapted
bases (cos/sin, even/odd polynomials) and the E/P operators come with.

Tensor convention: a rank-r field with parity p under a reflection R of
coordinate c satisfies  f(R x) = p · R_*(f)(x),  where R_* flips the sign of each
tensor index along c. Hence component (i1..ir) has scalar parity
p · (-1)^(number of indices equal to c). A true vector (velocity) has p = +1:
under x -> -x, u_x is odd and u_z even.
"""

class Parity(int):
    """+1 (even / symmetric) or -1 (odd / antisymmetric)."""

    def __new__(cls, value: int) -> Parity:
        if value not in (1, -1):
            raise SymmetryError(f"parity must be +1 or -1, got {value!r}", fix='Parity(+1) or Parity(-1)')
        return super().__new__(cls, value)

    def explain(self) -> str:
        return (f"Parity {int(self):+d} ({'even / symmetric' if self == 1 else 'odd / antisymmetric'}): "
                "f(R x) = p R_* f(x) under the declared reflection")

    def __mul__(self, other: Any) -> Any:  # type: ignore[override]
        if isinstance(other, int) and other in (1, -1):
            return Parity(int(self) * int(other))
        return int(self) * other

    __rmul__ = __mul__

    def __neg__(self) -> Parity:  # type: ignore[override]
        return Parity(-int(self))

    def __repr__(self) -> str:
        return "Parity(+1)" if self == 1 else "Parity(-1)"

    def __str__(self) -> str:
        return "even" if self == 1 else "odd"


EVEN, ODD = Parity(1), Parity(-1)


def _as_parity(p: Any) -> Parity:
    return p if isinstance(p, Parity) else Parity(int(p))


class _Z2Generator:
    """Common protocol of the generators of a symmetry subspace (Reflection, AzimuthalMFold).

    A field of parity p under a generator g satisfies f(g x) = p g_* f(x), p = +1 or -1 (a real
    character of the cyclic group generated by g). ``det`` is the determinant of g (-1 for a
    reflection: curls and cross products change sign; +1 for a rotation); ``flips(c)`` says whether
    g_* reverses the frame vector (and the derivative) along coordinate c; ``order`` is the order of
    the group generated by g (2 for a reflection, m0 for a rotation by 2 pi / m0)."""

    det = -1
    order = 2

    def flips(self, coord: Coordinate) -> bool:
        return coord == self.coord

class Reflection(_Z2Generator):
    """Reflection x_c -> 2*about - x_c. ``about=None`` means the centre of the interval.

    Curvilinear systems: the colatitude theta of S2/spherical coordinates (the equatorial
    reflection z -> -z, theta -> pi - theta, about pi/2 only; see ``EquatorialParity``) and the
    axis z of cylindrical coordinates. A reflection of the azimuth through a diameter
    (phi -> -phi) links m and -m and is not implemented (refused)."""

    def __init__(self, coord: Coordinate, about: float | None = None) -> None:
        if not isinstance(coord, Coordinate):
            raise TypeError("Reflection needs a Coordinate")
        cs = coord.system
        if cs.curvilinear:
            equatorial = cs.kind in ("S2", "Spherical") and coord.index == 1
            axial = cs.kind == "Cylindrical" and coord.index == 2
            if not (equatorial or axial):
                raise NotImplementedError(
                    f"reflection of {coord.name} in {cs.kind} coordinates: only the equatorial "
                    "reflection (colatitude theta -> pi - theta, S2/spherical) and the axial "
                    "reflection z -> -z (cylindrical) are implemented; a diameter reflection "
                    "phi -> -phi links m and -m (not implemented), use AzimuthalMFold for m classes")
            if equatorial and about is not None and abs(float(about) - math.pi / 2) > 1e-12:
                raise SymmetryError(f"equatorial reflection: theta -> pi - theta is about pi/2, "
                                    f"got about={about}")
        self.coord = coord
        self.about = None if about is None else float(about)

    @property
    def key(self) -> tuple:
        return ("Reflection", self.coord.key, self.about)

    def __eq__(self, other: object) -> bool:
        return isinstance(other, Reflection) and self.key == other.key

    def __hash__(self) -> int:
        return hash(self.key)

    def centre(self, basis: Basis) -> float:
        """Fixed point on ``basis``; checks that the reflection maps the domain to itself."""
        a, b = basis.bounds
        mid = 0.5 * (a + b)
        if self.about is None:
            return mid
        if basis.periodic:
            return self.about
        if abs(self.about - mid) > 1e-12 * max(1.0, abs(mid)):
            raise SymmetryError(
                f"{self}: interval {basis.bounds} is not symmetric about {self.about} "
                f"(its centre is {mid:g})")
        return self.about

    def mirror(self, pos: float, basis: Basis) -> float:
        x0 = self.centre(basis)
        m = 2 * x0 - pos
        if basis.periodic:
            a, L = basis.bounds[0], basis.length
            m = a + (m - a) % L
        return m

    def is_fixed(self, pos: float, basis: Basis) -> bool:
        m = self.mirror(pos, basis)
        if basis.periodic:
            L = basis.length
            d = abs(m - pos) % L
            return min(d, L - d) < 1e-12 * max(1.0, L)
        return abs(m - pos) < 1e-12 * max(1.0, abs(pos))



    @property
    def equatorial(self) -> bool:
        """True for the equatorial reflection theta -> pi - theta (S2/spherical coordinates)."""
        return self.coord.system.kind in ("S2", "Spherical") and self.coord.index == 1

    def __repr__(self) -> str:
        c = self.coord.name
        if self.equatorial:
            return f"Reflection({c} -> pi - {c}, equatorial)"
        return f"Reflection({c} -> {'-' if self.about in (None, 0.0) else f'{2 * self.about:g} - '}{c}" \
               f"{'' if self.about is not None else ', about centre'})"

class ShiftReflect(_Z2Generator):
    """Shift-and-reflect of a periodic pipe or cylinder: phi -> -phi composed with the
    half-period shift z -> z + Lz/2 of the axis,

        (u_phi, u_r, u_z)(phi, r, z) -> (-u_phi, u_r, u_z)(-phi, r, z + Lz/2)

    (the symmetry S of the pipe travelling waves: Wedin & Kerswell 2004, Pringle, Duguet &
    Kerswell 2009, Willis, Cvitanovic & Avila 2013). ``ShiftReflect(phi, z)`` on
    CylindricalCoordinates. A reflection (det -1: curls and cross products change sign) that
    reverses e_phi and d/dphi; the shift commutes with every z operator.

    On a stored real field (m >= 0) it pairs the coefficients (m, k_z) and (m, -k_z) through a
    complex conjugation: it is real-linear, not complex-linear, so it does not reduce the complex
    group systems; IVPs enforce it by a declared projection after every solve of the step
    (``timestep/symmetry_projection.py``: J read off the transforms, leak monitored and refused
    above its bound), together with the reduced m-fold classes. Cartesian shift-reflect (plane
    Couette S of Nagata) lives in ``conatus.ecs.subspace``."""

    det = -1
    order = 2

    def __init__(self, coord: Coordinate, shift: Coordinate) -> None:
        if not isinstance(coord, Coordinate) or not isinstance(shift, Coordinate):
            raise TypeError("ShiftReflect needs the azimuth and the axis Coordinates: ShiftReflect(phi, z)")
        cs = coord.system
        if not (cs.curvilinear and cs.kind == "Cylindrical" and coord.index == 0
                and shift.system is cs and shift.index == 2):
            raise NotImplementedError(
                "ShiftReflect(phi, z): azimuth and axis of CylindricalCoordinates (pipe, "
                "Taylor-Couette); the Cartesian shift-reflect of plane Couette is "
                "conatus.ecs.subspace (DiagonalSymmetryGroup)")
        self.coord, self.shift = coord, shift
        self.about = 0.0

    @property
    def key(self) -> tuple:
        """Hashable identity (generator kind and the two coordinates)."""
        return ("ShiftReflect", self.coord.key, self.shift.key)

    def __eq__(self, other: object) -> bool:
        return isinstance(other, ShiftReflect) and self.key == other.key

    def __hash__(self) -> int:
        return hash(self.key)

class AzimuthalMFold(_Z2Generator):
    """Rotation phi -> phi + 2 pi / m0 about the axis (disk, annulus, cylinder, sphere, shell, ball).

    Polar/spherical frame components are carried by the rotation (g_* e_r(x) = e_r(g x)), so a
    field of parity p has every component in the azimuthal class
    ``p = +1``: m = 0 mod m0 (m0-fold invariant); ``p = -1`` (m0 even): m = m0/2 mod m0
    (the rotation by 2 pi / m0 changes its sign). ``AzimuthalMFold(phi, 2)`` with EVEN / ODD
    keeps the even / odd azimuthal wavenumbers. A rotation preserves orientation (det = +1): curls
    and cross products keep their parity; no frame vector and no derivative is reversed."""

    det = +1

    def __init__(self, coord: Coordinate, m0: int = 2) -> None:
        if not isinstance(coord, Coordinate):
            raise TypeError("AzimuthalMFold needs the azimuth Coordinate")
        cs = coord.system
        if not (cs.curvilinear and cs.kind in ("Polar", "Cylindrical", "S2", "Spherical")
                and coord.index == 0):
            raise NotImplementedError(
                f"AzimuthalMFold({coord.name}): the rotation acts on the azimuth phi of polar, "
                "cylindrical, S2 or spherical coordinates (in Cartesian coordinates use "
                "Reflection / shift symmetries)")
        if int(m0) != m0 or int(m0) < 2:
            raise SymmetryError(f"AzimuthalMFold: m0 must be an integer >= 2, got {m0!r}")
        self.coord, self.m0 = coord, int(m0)
        self.about = None

    @property  # type: ignore[override]
    def order(self) -> int:  # noqa: D102
        return self.m0

        return False

        return ("AzimuthalMFold", self.coord.key, self.m0)

        return isinstance(other, AzimuthalMFold) and self.key == other.key

    def __hash__(self) -> int:
        return hash(self.key)

    def residue(self, p: Parity) -> int:
        """Azimuthal class of parity p: kept wavenumbers are m = residue (mod m0)."""
        if p == 1:
            return 0
        if self.m0 % 2:
            raise SymmetryError(f"{self!r}: an odd m0 has no sign-changing real character "
                                "(parity -1 needs m0 even)")
        return self.m0 // 2

    def keeps(self, m: int, p: Parity) -> bool:
        """Is azimuthal wavenumber m in the class of parity p?"""
        return (int(m) - self.residue(p)) % self.m0 == 0

    def centre(self, basis: Basis) -> float:
        """Validates the azimuthal basis (Fourier over 2 pi, size a multiple of m0)."""
        if not basis.periodic or abs(basis.length - 2 * math.pi) > 1e-12:
            raise SymmetryError(f"{self!r}: the azimuthal basis must be periodic over 2 pi")
        if basis.size % self.m0:
            raise SymmetryError(f"{self!r}: Nphi = {basis.size} is not a multiple of m0 = {self.m0}")
        return 0.0

        return (pos + 2 * math.pi / self.m0) % (2 * math.pi)

def EquatorialParity(coordsys: Any) -> Reflection:
    """Equatorial reflection z -> -z (theta -> pi - theta) of S2 or spherical coordinates: the
    parity (-1)^(l+m) of Y_l^m (spin components: ``u_+ <-> -u_-``).
    Returns ``Reflection(theta, about=pi/2)``; accepts the coordinate system or one of its
    coordinates."""
    from .coords import CoordinateSystem
    if isinstance(coordsys, Coordinate):
        coordsys = coordsys.system
    if not isinstance(coordsys, CoordinateSystem) or coordsys.kind not in ("S2", "Spherical"):
        raise NotImplementedError("EquatorialParity needs S2Coordinates or SphericalCoordinates "
                                  "(Cartesian and cylindrical systems: Reflection(z))")
    return Reflection(coordsys.coords[1], about=math.pi / 2)

def _normalize_field_symmetry(sym: Any, generators: tuple[_Z2Generator, ...] | None = None
                              ) -> tuple[tuple[_Z2Generator, Parity], ...]:
    if sym is None:
        return ()
    if isinstance(sym, (Parity, int)):
        if generators is None or len(generators) != 1:
            raise SymmetryError("a bare Parity is only unambiguous with a single reflection; "
                                "give {Reflection: Parity}")
        return ((generators[0], _as_parity(sym)),)
    if isinstance(sym, Mapping):
        out = []
        for r, p in sym.items():
            if not isinstance(r, _Z2Generator):
                raise NotImplementedError(f"symmetry {r!r}: only Reflection and AzimuthalMFold "
                                          "generators are supported")
            out.append((r, _as_parity(p)))
        return tuple(sorted(out, key=lambda rp: repr(rp[0].key)))
    raise TypeError(f"cannot interpret symmetry declaration {sym!r}")

class SymmetrySubspace:
    """Invariant subspace of a group generated by reflections, with field parities.

    ``SymmetrySubspace(Rx, u=EVEN, T=ODD)`` or, with several generators,
    ``SymmetrySubspace(Rx, Rz, u={Rx: EVEN, Rz: ODD}, ...)``. Parities may also be
    declared on the Field itself (``Field(..., symmetry={Rx: ODD})``).
    """

    def __init__(self, *generators: Any, **parities: Any) -> None:
        for g in generators:
            if not isinstance(g, _Z2Generator):
                raise NotImplementedError(f"generator {g!r}: only Reflection and AzimuthalMFold "
                                          "generators are supported")
        if len(set(generators)) != len(generators):
            raise SymmetryError("duplicate generators")
        # ShiftReflect(phi, z) and AzimuthalMFold(phi, m0) both act on phi and generate a
        # dihedral group (S R S = R^-1): their common fixed subspace is well defined (allowed)
        coords = [g.coord for g in generators if not isinstance(g, ShiftReflect)]
        if len(set(coords)) != len(coords):
            raise SymmetryError("two generators acting on the same coordinate: two reflections "
                                "compose to a translation, two rotations to one rotation; not a "
                                "new independent factor")
        self.generators: tuple[_Z2Generator, ...] = tuple(generators)
        self.parities: dict[str, dict[_Z2Generator, Parity]] = {
            name: dict(_normalize_field_symmetry(p, self.generators)) for name, p in parities.items()}
        for _name, ps in self.parities.items():      # a sign under an odd rotation has no meaning
            for g, p in ps.items():
                if isinstance(g, AzimuthalMFold):
                    g.residue(p)

    @property
    def order(self) -> int:
        """Order of the group (product of the generators' orders; 2^k for k reflections)."""
        out = 1
        for g in self.generators:
            out *= g.order
        return out

    def parity(self, field: Field, R: _Z2Generator) -> Parity:
        declared = dict(field.symmetry).get(R)
        here = self.parities.get(field.name, {}).get(R)
        if declared is not None and here is not None and declared != here:
            raise SymmetryError(f"conflicting parities for {field.name} under {R!r}: "
                                f"{declared} on the Field, {here} in the subspace")
        p = declared if declared is not None else here
        if p is None:
            raise SymmetryError(f"parity of field '{field.name}' under {R!r} is not declared")
        return p

class ParityLookup:
    """Callable field -> parity under a reflection (from the subspace and Field declarations)."""

    def __init__(self, sub: SymmetrySubspace | None) -> None:
        self.sub = sub

    def __call__(self, field: Field, R: _Z2Generator) -> Parity:
        if self.sub is not None:
            return self.sub.parity(field, R)
        p = dict(field.symmetry).get(R)
        if p is None:
            raise SymmetryError(f"parity of field '{field.name}' under {R!r} is not declared")
        return p

def component_parity(p: Parity, tensorsig: tuple, index: tuple[int, ...], R: _Z2Generator) -> Parity:
    """Scalar parity of component ``index`` of a tensor with tensor parity ``p`` (one sign per
    index along a frame vector the generator reverses; none for a rotation)."""
    out = p
    for cs, i in zip(tensorsig, index, strict=True):
        if R.flips(cs.coords[i]):
            out = -out
    return out


def _jacobi_symmetric(b: Basis) -> bool:
    return isinstance(b, Jacobi) and b.a == b.b


def parity_of(e: Expr, R: _Z2Generator, lookup: ParityLookup | None = None) -> Parity | None:
    """Tensor parity of ``e`` under ``R``; None means 'identically zero' (compatible with both)."""
    from . import expr as E

    lookup = lookup or ParityLookup(None)
    cache: dict = {}

    def rec(x: Expr) -> Parity | None:
        if x in cache:
            return cache[x]
        r = _rule(x)
        cache[x] = r
        return r

    def _rule(x: Expr) -> Parity | None:
        if isinstance(x, E.Field):
            if x.domain.has(R.coord):
                R.centre(x.domain.basis(R.coord))  # validates the reflection on this domain
            return lookup(x, R)
        if isinstance(x, E.Constant):
            return None if x.is_zero() else EVEN
        if isinstance(x, E.Parameter):
            return EVEN
        if isinstance(x, E.UnitVector):
            return ODD if R.flips(x.coordsys.coords[x.index]) else EVEN
        if isinstance(x, E.Add):
            ps = [(t, rec(t)) for t in x.args]
            known = [(t, p) for t, p in ps if p is not None]
            if not known:
                return None
            t0, p0 = known[0]
            for t, p in known[1:]:
                if p != p0:
                    raise SymmetryBreakingError(
                        f"term '{t}' is {p} under {R!r} but '{t0}' is {p0}: "
                        "the sum breaks the declared symmetry")
            return p0
        if isinstance(x, (E.Mul, E.Dot, E.Outer)):
            out = EVEN
            for a in x.args:
                p = rec(a)
                if p is None:
                    return None
                out = out * p
            return out
        if isinstance(x, E.Power):
            p = rec(x.args[0])
            if p is None or p == EVEN:
                return p
            ex = x.exponent
            if float(ex).is_integer():
                return EVEN if int(ex) % 2 == 0 else ODD
            raise SymmetryBreakingError(f"'{x}': non-integer power of an odd quantity")
        if isinstance(x, E.UnaryFunction):
            p = rec(x.args[0])
            if p is None:                       # f(0): zero, a constant, or undefined
                f0 = x.value_at_zero
                if f0 is None:
                    raise SymmetryBreakingError(f"'{x}': {x.func}(0) is undefined")
                return None if f0 == 0 else EVEN
            if p == EVEN:
                return EVEN
            if x.parity_kind == "odd":
                return ODD
            if x.parity_kind == "even":
                return EVEN
            raise SymmetryBreakingError(
                f"'{x}': {x.func}() of an odd quantity has no parity under {R!r}")
        if isinstance(x, (E.Conj, E.TimeDerivative, E.Gradient, E.Divergence, E.Laplacian)):
            return rec(x.args[0])
        if isinstance(x, E.Curl):
            p = rec(x.args[0])
            return None if p is None else p * R.det  # pseudo-tensor: sign det(R)
        if isinstance(x, E.Cross):
            pa, pb = rec(x.args[0]), rec(x.args[1])
            if pa is None or pb is None:
                return None
            return pa * pb * R.det                   # a x b is a pseudo-vector: sign det(R)
        if isinstance(x, E.Component):
            p = rec(x.args[0])
            if p is None:
                return None
            return -p if R.flips(x.args[0].tensorsig[0].coords[x.index]) else p
        if isinstance(x, E.Differentiate):
            p = rec(x.args[0])
            if p is None:
                return None
            return -p if R.flips(x.coord) else p
        if isinstance(x, E.Integrate):
            p = rec(x.args[0])
            if p is None:
                return None
            if R.coord in x.coords:
                return EVEN if p == EVEN else None  # odd integrand over a symmetric domain: 0
            return p
        if isinstance(x, E.Interpolate):
            p = rec(x.args[0])
            if p is None or x.coord != R.coord:
                return p
            basis = x.args[0].domain.basis(x.coord)
            if R.is_fixed(x.position, basis):
                return EVEN if p == EVEN else None  # odd quantity vanishes on the mirror plane
            return p  # trace at x0; its image is the trace at the mirror point (see BC check)
        if isinstance(x, E.Lift):
            p = rec(x.args[0])
            if p is None or x.basis.coord != R.coord:
                return p
            if not _jacobi_symmetric(x.basis):
                raise SymmetryBreakingError(f"'{x}': Jacobi(a != b) lift polynomial has no parity")
            k = x.basis.size + x.n
            return p if k % 2 == 0 else -p
        raise NotImplementedError(f"parity rule for {type(x).__name__}")

    return rec(e)
